
use crate::{lexer::Lexer, token::Token};
use ast::{Block, Expression, Literal, Precedence, Program, Statement, precedence_of};

#[cfg(test)]
mod tests;

pub mod ast;

pub struct Parser {
    lexer: Lexer,
    current: Token,
    peek_token: Token,
    errors: Vec<String>,
}

impl Parser {
    pub fn new(lexer: Lexer) -> Self {
        let mut parser = Parser {
            lexer,
            errors: vec![],
            current: Token::EOF,
            peek_token: Token::EOF,
        };

        parser.advance(); // Initialize current token
        parser.advance(); // Initialize peek token
        parser
    }

    fn advance(&mut self) -> &Token {
        self.current = self.peek_token.clone();
        self.peek_token = self.lexer.next_token();
        &self.current
    }

    fn expect_identifier(&mut self) -> String {
        if let Token::IDENTIFIER(name) = &self.current {
            name.clone()
        } else {
            self.errors
                .push(format!("Expected identifier but got {:?}", self.current));
            "".into()
        }
    }

    fn expect_string(&mut self) -> String {
        if let Token::STRING(value) = self.advance() {
            value.clone()
        } else {
            self.errors.push(format!(
                "Expected string literal but got {:?}",
                self.current
            ));
            "".into()
        }
    }

    fn expect_number(&mut self) -> i64 {
        if let Token::NUMBER(n) = self.advance() {
            *n
        } else {
            self.errors.push(format!(
                "Expected number literal but got {:?}",
                self.current
            ));
            0
        }
    }

    pub fn parse(&mut self) -> Result<Program, Vec<String>> {
        let mut body = vec![];
        while !matches!(self.current, Token::EOF) {
            body.push(self.parse_statement());
            self.advance();
        }

        if self.errors.is_empty() {
            Ok(Program { body })
        } else {
            Err(self.errors.clone())
        }
    }

    fn parse_statement(&mut self) -> Statement {
        if let Some(expr) = self.try_parse_expression_statement() {
            return Statement::ExpressionStatement(expr);
        }
        if let Token::KEYWORD(k1) = &self.current {
            if k1 == "circle" && matches!(self.peek_token, Token::KEYWORD(ref k2) if k2 == "back") {
                self.advance(); // consume 'back'
                return Statement::Continue;
            }
            if k1 == "punt" {
                self.advance(); // consume 'pivotaway'
                return Statement::Break;
            }
        }

        match &self.current {
            Token::KEYWORD(k) if k == "actualize" => self.parse_variable_declaration(),
            Token::IDENTIFIER(_) => self.parse_assignment(),
            Token::KEYWORD(k) if k == "touch" => self.parse_print(),
            Token::KEYWORD(k) if k == "pivot" => self.parse_conditional(),
            Token::KEYWORD(k) if k == "optimize" => self.parse_loop(),
            _ => {
                self.errors
                    .push(format!("Unexpected token: {:?}", self.current));
                Statement::Print {
                    value: Expression::Literal(Literal::String(String::from("error"))),
                }
            }
        }
    }

    fn parse_variable_declaration(&mut self) -> Statement {
        self.advance(); // actualize
        let name = self.expect_identifier();
        self.expect_peek(&Token::KEYWORD(String::from("align")));
        let value = self.parse_expression();
        Statement::VariableDeclaration { name, value }
    }

    fn parse_assignment(&mut self) -> Statement {
        let name = self.expect_identifier();
        self.expect_peek(&Token::KEYWORD(String::from("align")));
        let value = self.parse_expression();
        Statement::Assignment { name, value }
    }

    fn parse_print(&mut self) -> Statement {
        self.advance(); // touch
        self.expect_peek(&Token::KEYWORD(String::from("with")));
        let value = self.parse_expression();
        Statement::Print { value }
    }

    fn parse_block_statement(&mut self) -> Block {
        let mut statements = vec![];
        while !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
            let stmt = self.parse_statement();
            statements.push(stmt);
            if !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
                self.advance();
            }
        }
        Block { statements }
    }

    fn parse_conditional(&mut self) -> Statement {
        self.advance(); // pivot
        let condition = self.parse_expression();
        let mut then_branch = Block { statements: vec![] };
        let mut else_branch = None;

        while !matches!(&self.current, Token::KEYWORD(k) if k == "else" || k == "end") {
            let stmt = self.parse_statement();
            // let stmt = self.parse_block_statement();
            // then_branch = stmt;
            then_branch.statements.push(stmt);
            if !matches!(&self.current, Token::KEYWORD(k) if k == "else" || k == "end") {
                self.advance();
            }
        }

        if matches!(&self.current, Token::KEYWORD(k) if k == "else") {
            self.advance(); // else
            let mut else_statements = Block { statements: vec![] };
            while !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
                let stmt = self.parse_statement();
                // let stmt = self.parse_block_statement();
                else_statements.statements.push(stmt);
                if !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
                    self.advance();
                }
            }
            else_branch = Some(else_statements);
        }

        if matches!(&self.current, Token::KEYWORD(k) if k == "end") {
            self.advance(); // ✅ move past end so outer loop continues cleanly
        } else {
            self.errors
                .push("Expected 'end' to close pivot block".into());
        }

        Statement::Conditional {
            condition,
            then_branch,
            else_branch,
        }
    }

    fn parse_loop(&mut self) -> Statement {
        self.advance(); // optimize
        if !matches!(&self.current, Token::KEYWORD(k) if k == "while") {
            self.errors.push("Expected 'while' after 'optimize'".into());
        } else {
            self.advance();
        }

        let condition = self.parse_expression();
        let mut body = Block { statements: vec![] };

        while !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
            let stmt = self.parse_statement();
            match stmt {
                Statement::Continue | Statement::Break => body.statements.push(stmt),
                _ => body.statements.push(stmt),
            }
            if !matches!(&self.current, Token::KEYWORD(k) if k == "end") {
                self.advance();
            }
        }

        if matches!(&self.current, Token::KEYWORD(k) if k == "end") {
            self.advance();
        } else {
            self.errors
                .push(String::from("Expected 'end' to close loop block"));
        }

        Statement::Loop { condition, body }
    }

    fn parse_expression(&mut self) -> Expression {
        self.parse_expression_precedence(Precedence::Lowest)
    }

    fn parse_expression_precedence(&mut self, precedence: Precedence) -> Expression {
        let mut left = match &self.current {
            Token::IDENTIFIER(name) => {
                let expr = Expression::Identifier(name.clone());
                self.advance();
                expr
            }
            Token::KEYWORD(_) => {
                self.advance();
                self.parse_expression()
            }
            Token::NUMBER(n) => {
                let expr = Expression::Literal(Literal::Number(*n));
                self.advance();
                expr
            }
            Token::STRING(s) => {
                let expr = Expression::Literal(Literal::String(s.clone()));
                self.advance();
                expr
            }
            Token::BOOLEAN(b) => {
                let expr = Expression::Literal(Literal::Boolean(*b));
                self.advance();
                expr
            }
            _ => {
                self.errors.push(format!(
                    "Unexpected token in expression: {:?}",
                    self.current
                ));
                self.advance();
                return Expression::Literal(Literal::String(String::from("error")));
            }
        };

        while let Token::OPERATOR(op) = &self.current {
            let op_prec = precedence_of(op);
            if op_prec <= precedence {
                break;
            }
            let operator = op.clone();
            self.advance();
            let right = self.parse_expression_precedence(op_prec);
            left = Expression::Binary {
                left: Box::new(left),
                operator,
                right: Box::new(right),
            };
        }

        left
    }

    fn try_parse_expression_statement(&mut self) -> Option<Expression> {
        match self.current {
            Token::IDENTIFIER(_) | Token::NUMBER(_) | Token::STRING(_) | Token::BOOLEAN(_) => {
                Some(self.parse_expression())
            }
            _ => None,
        }
    }

    fn expect_peek(&mut self, expected: &Token) -> Option<&Token> {
        if &self.peek_token == expected {
            self.advance();
            Some(&self.current)
        } else {
            self.peek_error(expected);
            None
        }
    }

    fn peek_error(&mut self, expected: &Token) {
        let msg = format!(
            "Expected token {:?}, but got {:?}",
            expected, self.peek_token
        );
        self.errors.push(msg);
    }
}
